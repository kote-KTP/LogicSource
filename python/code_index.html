def prepare_plot_data(values):
    if values is None:
        return values
    if isinstance(values, pd.DataFrame):
        if values.shape[1] == 0:
            return pd.Series(dtype="float64")
        values = values.iloc[:, 0]
    
    series = pd.Series(values)
    if len(series) == 0:
        return series
    return series
